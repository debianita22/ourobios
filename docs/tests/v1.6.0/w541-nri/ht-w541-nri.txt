## rcba.py
import mmap, os, struct, sys
# usage: rcba.py [OFF=VAL ...]  (prints 0x38f0/0x38f4/0x38f8 after the writes)
fd=os.open("/dev/mem",os.O_RDWR|os.O_SYNC); m=mmap.mmap(fd,0x1000,mmap.MAP_SHARED,mmap.PROT_READ|mmap.PROT_WRITE,offset=0xfed1f000)
for a in sys.argv[1:]:
    o,v=a.split("="); o=int(o,16); v=int(v,16); struct.pack_into("<I",m,o-0x3000,v)
print(" ".join(f"0x{o:04x}=0x{struct.unpack_from('<I',m,o-0x3000)[0]:08x}" for o in (0x38f0,0x38f4,0x38f8)))

## ouro-exp.sh (the reboot chain, run once per boot by a oneshot service)
#!/bin/bash
# Reboot chain for the W541 roadmap data (Hyper-Threading via SOFT_RESET_DATA,
# DRAM clear on MOR, cold boot). Run once per boot by ouro-exp.service.
OUT=/home/domenico/ourobios-test-v1.6.0/w541-nri/dev
REPO=/home/domenico/ourobios-next
ST=/var/lib/ouro-exp/step
MORV=/sys/firmware/efi/efivars/MemoryOverwriteRequestControl-e20939be-32d4-41be-a150-897f85d49829
step=$(cat $ST 2>/dev/null || echo none)
sleep 45
modprobe msr
collect() {  # $1 = name
  local d=$OUT/$1; mkdir -p $d
  {
    echo "# $1  $(date -Is)  boot_id=$(cat /proc/sys/kernel/random/boot_id)"
    echo "## SOFT_RESET (RCBA 0x38f0 lock, 0x38f4 ctrl, 0x38f8 data)"; python3 -I /usr/local/sbin/ouro-rcba.py
    echo "## CPUs"; echo "online=$(cat /sys/devices/system/cpu/online) offline=$(cat /sys/devices/system/cpu/offline) present=$(cat /sys/devices/system/cpu/present)"
    for c in $(seq 0 7); do echo "cpu$c MSR_CORE_THREAD_COUNT(0x35)=$(rdmsr -p$c 0x35 2>&1)"; done
    lscpu | grep -E "^CPU\(s\)|On-line|Off-line|Thread|Core"
    echo "## MOR"; od -An -tx1 $MORV
    echo "## kernel: CPU bring-up and USB SS2"; journalctl -k -b --no-pager | grep -E "smpboot|failed to report|Error taking|TSC ADJUST|Allowing|APIC" | head -30
    echo "usb4-port2 Cannot enable: $(journalctl -k -b --no-pager | grep -c 'usb4-port2: Cannot enable'), first: $(journalctl -k -b --no-pager | grep -m1 'Cannot enable' | cut -c1-15)"
  } > $d/summary.txt 2>&1
  echo on > /sys/bus/pci/devices/0000:00:14.0/power/control; mount | grep -q debugfs || mount -t debugfs none /sys/kernel/debug
  sleep 1; for p in /sys/kernel/debug/usb/xhci/0000:00:14.0/ports/port*; do echo "$(basename $p) $(cat $p/portsc)"; done > $d/xhci-portsc.txt
  echo auto > /sys/bus/pci/devices/0000:00:14.0/power/control
  cbmem -1 > $d/cbmem-1.txt 2>&1; cbmem -t > $d/cbmem-t.txt 2>&1
  python3 -I /usr/local/sbin/ouro-pchregs.py > $d/pchregs.txt 2>&1
  python3 -I $REPO/tools/hw-report.py --markdown $d/hw-report.md > $d/hw-report.txt 2>&1
  python3 -I $REPO/tools/elog.py > $d/elog.txt 2>&1
  journalctl -k -b --no-pager > $d/dmesg.txt 2>&1
  chown -R domenico:domenico $OUT
}
case $step in
  B)  # after an ACPI reset (CF9=6) with SOFT_RESET_CTRL=1, DATA=1
      collect B-reboot-ht-off
      # back on through the same path: DATA=0, CTRL stays 1; and a MOR request for the DRAM clear timing
      python3 -I /usr/local/sbin/ouro-rcba.py 38f8=0 38f4=1 >> $OUT/B-reboot-ht-off/summary.txt
      chattr -i $MORV 2>/dev/null; printf '\x07\x00\x00\x00\x01' > $MORV; echo "MOR set: $(od -An -tx1 $MORV)" >> $OUT/B-reboot-ht-off/summary.txt
      echo C > $ST; sync; systemctl reboot ;;
  C)  collect C-reboot-ht-on-mor
      # does the bit survive S5? set it again, then a cold boot by the RTC
      python3 -I /usr/local/sbin/ouro-rcba.py 38f8=1 38f4=1 >> $OUT/C-reboot-ht-on-mor/summary.txt
      echo D > $ST; sync
      rtcwake -m off -s 120 ;;
  D)  collect D-cold-boot
      if [ "$(cat /sys/devices/system/cpu/online)" != "0-7" ]; then
        python3 -I /usr/local/sbin/ouro-rcba.py 38f8=0 38f4=1 >> $OUT/D-cold-boot/summary.txt
        echo E > $ST; sync; systemctl reboot
      else
        python3 -I /usr/local/sbin/ouro-rcba.py 38f4=0 38f8=0 >> $OUT/D-cold-boot/summary.txt
        echo done > $ST; systemctl disable ouro-exp.service
      fi ;;
  E)  collect E-reboot-ht-restored
      python3 -I /usr/local/sbin/ouro-rcba.py 38f4=0 38f8=0 >> $OUT/E-reboot-ht-restored/summary.txt
      echo done > $ST; systemctl disable ouro-exp.service ;;
esac
