## watch.sh
#!/bin/bash
# logs each USB attach/detach on the xHCI with the port, speed, PORTSC and the firmware's _UPC/_PLD
D=/home/domenico/ourobios-test-v1.6.0/w541-nri/diag/portmap; LOG=$D/events.txt
mount | grep -q debugfs || mount -t debugfs none /sys/kernel/debug
X=/sys/kernel/debug/usb/xhci/0000:00:14.0/ports
echo on > /sys/bus/pci/devices/0000:00:14.0/power/control
pld() { local p=$1; local s=/sys/bus/usb/devices/$p; echo "connect_type=$(cat $s/connect_type 2>/dev/null) _PLD=$(cat $s/physical_location/panel 2>/dev/null)/$(cat $s/physical_location/vertical_position 2>/dev/null)/$(cat $s/physical_location/horizontal_position 2>/dev/null) peer=$(basename "$(readlink $s/peer 2>/dev/null)" 2>/dev/null)"; }
echo "# start $(date -Is)" >> $LOG
journalctl -k -f -n 0 --no-pager | grep --line-buffered -E "usb [24]-[0-9]+: (new .*USB device|USB disconnect)" | while read -r line; do
  sleep 1.5
  dev=$(echo "$line" | grep -oE "usb [24]-[0-9]+" | cut -d' ' -f2); bus=${dev%%-*}; port=${dev##*-}
  {
    echo "== $(date +%T) $line" | sed 's/Wurtz kernel: //'
    echo "   root port: usb${bus}-port${port}  $(pld 4-0:1.0/usb4-port$port 2>/dev/null | grep -q . && true)"
    [ $bus = 4 ] && echo "   USB3 SS$port: $(pld 4-0:1.0/usb4-port$port)" || echo "   USB2 HS$port: $(pld 2-0:1.0/usb2-port$port)"
    if echo "$line" | grep -q new; then
      echo "   device: $(cat /sys/bus/usb/devices/$dev/manufacturer 2>/dev/null) $(cat /sys/bus/usb/devices/$dev/product 2>/dev/null) $(cat /sys/bus/usb/devices/$dev/idVendor 2>/dev/null):$(cat /sys/bus/usb/devices/$dev/idProduct 2>/dev/null) speed=$(cat /sys/bus/usb/devices/$dev/speed 2>/dev/null) Mb/s"
    fi
    echo -n "   non-idle PORTSC:"; for p in $X/port*; do v=$(cut -d' ' -f1 $p/portsc); case $v in 0x0a0002a0|0x000002a0) ;; *) echo -n " $(basename $p)=$(cat $p/portsc)";; esac; done; echo
  } >> $LOG
done
